Linear SVM: Effect of the C Parameter

Copy the complete code below into one Jupyter Notebook cell and run it. The program will generate the comparison figure automatically.

What you should observe: small C is more tolerant of margin violations and usually produces stronger regularization; large C penalizes violations more strongly. Compare the margin width, number of support vectors, and training accuracy across the three plots. 
"""
Simple Linear SVM Demo
3 functions: make_data, plot_svm, compare_C
"""

import numpy as np
import matplotlib.pyplot as plt
from sklearn.svm import SVC

np.random.seed(42)

# ============================================
# FUNCTION 1: Create a Simple Dataset
# ============================================
def make_data():
    """
    Create two overlapping classes of 2D data.
    Each observation has two features.
    """
    # Class 0: cluster toward the bottom-left
    X1 = np.random.randn(50, 2) + [-1, -1]

    # Class 1: cluster toward the top-right
    X2 = np.random.randn(50, 2) + [1, 1]

    # Combine the two groups
    X = np.vstack([X1, X2])

    # First 50 labels are 0; next 50 labels are 1
    y = np.array([0] * 50 + [1] * 50)

    return X, y


# ============================================
# FUNCTION 2: Train and Plot One SVM
# ============================================
def plot_svm(X, y, C, ax):
    """Train a Linear SVM and visualize the boundary, margins, and support vectors."""

    # Train a linear SVM
    # C controls how strongly margin violations/errors are penalized.
    svm = SVC(kernel='linear', C=C)
    svm.fit(X, y)

    # Linear SVM decision boundary:
    # w1*x1 + w2*x2 + b = 0
    w = svm.coef_[0]
    b = svm.intercept_[0]

    # x-values used to draw the lines
    x_points = np.linspace(-5, 5, 100)

    # Solve the SVM equations for x2 (vertical axis):
    #
    # Decision boundary: w·x + b = 0
    # Upper margin:      w·x + b = +1
    # Lower margin:      w·x + b = -1
    y_boundary = -(w[0] * x_points + b) / w[1]
    y_margin_up = -(w[0] * x_points + b - 1) / w[1]
    y_margin_down = -(w[0] * x_points + b + 1) / w[1]

    # Plot decision boundary and margins
    ax.plot(x_points, y_boundary, 'k-', linewidth=3, label='Decision Boundary')
    ax.plot(x_points, y_margin_up, 'b--', linewidth=2, label='Margin')
    ax.plot(x_points, y_margin_down, 'r--', linewidth=2)

    # Plot the two classes
    ax.scatter(X[y == 0, 0], X[y == 0, 1],
               c='blue', s=50, label='Class 0')

    ax.scatter(X[y == 1, 0], X[y == 1, 1],
               c='red', s=50, label='Class 1')

    # Highlight support vectors with large green circles
    ax.scatter(
        svm.support_vectors_[:, 0],
        svm.support_vectors_[:, 1],
        s=200,
        facecolors='none',
        edgecolors='green',
        linewidths=3,
        label='Support Vectors'
    )

    # Full distance between the two canonical margin lines
    margin_width = 2 / np.linalg.norm(w)

    # Number of support vectors
    n_sv = len(svm.support_vectors_)

    # Training accuracy: evaluated on the SAME data used for training
    train_acc = svm.score(X, y)

    # Display statistics in the title
    ax.set_title(
        f'C={C} | Margin={margin_width:.2f} | '
        f'SVs={n_sv} | Train Acc={train_acc:.2%}',
        fontsize=12,
        fontweight='bold'
    )

    ax.legend()
    ax.grid(True, alpha=0.3)
    ax.set_xlim(-5, 5)
    ax.set_ylim(-5, 5)


# ============================================
# FUNCTION 3: Compare Different C Values
# ============================================
def compare_C():
    """Compare three C values using exactly the same dataset."""

    # Create the data ONCE
    X, y = make_data()

    # Small, medium, and large C
    C_values = [0.01, 1, 100]

    # Create three side-by-side plots
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))

    # Train and plot one SVM for each C
    for i, C in enumerate(C_values):
        plot_svm(X, y, C, axes[i])

    plt.suptitle(
        'Linear SVM: Effect of C Parameter',
        fontsize=16,
        fontweight='bold'
    )

    plt.tight_layout()

    # Save the comparison figure
    plt.savefig(
        'simple_svm_demo.png',
        dpi=150,
        bbox_inches='tight'
    )

    print("Saved: simple_svm_demo.png")
    plt.show()


# ============================================
# BONUS: Display One C Value
# ============================================
def single_plot(C=1):
    """Train and display one SVM for a chosen C value."""

    X, y = make_data()

    fig, ax = plt.subplots(figsize=(8, 6))
    plot_svm(X, y, C, ax)

    plt.tight_layout()
    plt.show()


# ============================================
# RUN THE PROGRAM
# ============================================
if __name__ == "__main__":
    print("Running Simple SVM Demo...")

    # Compare C = 0.01, 1, and 100
    compare_C()

    print("\nDone! Check 'simple_svm_demo.png'")

    # Try one additional C value
    single_plot(C=0.1)


# ============================================
# MAIN IDEA OF C
# ============================================

# C = 0.01
#   Small penalty for violations/errors
#   Stronger regularization
#   More tolerant of training violations
#   Often gives a wider/softer margin
#
# C = 1
#   Moderate penalty
#   Balance between margin size and violations
#
# C = 100
#   Large penalty for violations/errors
#   Weaker regularization
#   Tries harder to fit the training data
#
# Note:
# A very large finite C is not literally the same
# as a mathematical hard-margin SVM.
Workflow:
Create overlapping data → Train Linear SVM → Calculate decision boundary and margins → Highlight support vectors → Compare C = 0.01, 1, 100 → Display and save figure
